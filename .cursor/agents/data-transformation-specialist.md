# Data Transformation Specialist

## Responsibilities

- Implement normalizers in `src/lib/normalizers/`
- Implement serializers in `src/lib/serializers/`
- Ensure that only used in API routes when there is communication with external APIs
- Keep documentation in `src/lib/TRANSFORMATIONS.md`

## When to Use

**Normalizers and Serializers are only used in API routes (`src/pages/api/`) when the route calls an external API.**

- **Normalizers**: External API (verbose) → Internal (short names)
- **Serializers**: Internal (short names) → External API (verbose)

## Normalizers (External → Internal)

### Example: User Normalizer

```tsx
// src/lib/normalizers/user-normalizer.ts
import type { User } from "@/types/user";

export const normalizeUser = (external: Record<string, unknown>): User => {
  return {
    id: external._id as string,
    name: external.fullName as string,
    email: external.emailAddress as string,
    role: (external.userRole as string).toLowerCase() as User["role"],
  };
};

export const normalizeUserList = (
  externalList: Record<string, unknown>[]
): User[] => {
  return externalList.map(normalizeUser);
};
```

## Serializers (Internal → External)

### Example: User Serializer

```tsx
// src/lib/serializers/user-serializer.ts
import type { User, CreateUserBody } from "@/types/user";

const serializeRole = (role: User["role"]): string => {
  return role.toLowerCase();
};

export const serializeUser = (
  user: User
): Record<string, unknown> => {
  return {
    _id: user.id,
    fullName: user.name,
    emailAddress: user.email,
    userRole: serializeRole(user.role),
  };
};

export const serializeCreateUser = (
  data: CreateUserBody
): Record<string, unknown> => {
  return {
    fullName: data.name,
    emailAddress: data.email,
    userRole: serializeRole(data.role),
    // ... (exclude id since it is generated by the API)
  };
};

export const serializeUpdateUser = (
  data: Partial<User>
): Record<string, unknown> => {
  const serialized: Record<string, unknown> = {};

  if (data.name !== undefined) serialized.fullName = data.name;
  if (data.role !== undefined) {
    serialized.userRole = serializeRole(data.role);
  }
  // ... only serialize present fields

  return serialized;
};
```

## Usage in API Routes

```tsx
// src/pages/api/users.ts
import type { NextApiRequest, NextApiResponse } from "next";
import { HTTP_RESPONSE_MESSAGE, type HttpResponses } from "@/types/http-responses";
import { normalizeUser, normalizeUserList } from "@/lib/normalizers";
import { serializeCreateUser } from "@/lib/serializers";

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method === "GET") {
    const response = await fetch(`${process.env.EXTERNAL_API_URL}/users`);
    const externalData = await response.json();

    // ✅ Normalize: External → Internal
    const users = normalizeUserList(externalData.data);

    return res.status(200).json({
      message: HTTP_RESPONSE_MESSAGE.SUCCESS,
      data: { users },
      status: 200,
    });
  }

  if (req.method === "POST") {
    const body = req.body as CreateUserBody;

    // ✅ Serialize: Internal → External
    const serializedData = serializeCreateUser(body);

    const response = await fetch(`${process.env.EXTERNAL_API_URL}/users`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(serializedData),
    });

    const externalData = await response.json();

    // ✅ Normalize response: External → Internal
    const user = normalizeUser(externalData);

    return res.status(201).json({
      message: HTTP_RESPONSE_MESSAGE.SUCCESS,
      data: { user },
      status: 201,
    });
  }
}
```

## Field Mappings (Example)

| Internal (short) | External API (verbose) | Transformation |
| ------------------ | ------------------------ | ---------------- |
| `id` | `_id` | Direct mapping |
| `name` | `fullName` | Direct mapping |
| `email` | `emailAddress` | Direct mapping |
| `role: "admin"` | `userRole: "admin"` | Direct mapping |

## Important Rules

1. **Location**: `src/lib/normalizers/` and `src/lib/serializers/`
2. **Only in API routes**: When the route calls an external API
3. **Naming**: kebab-case for files (`employee-normalizer.ts`)
4. **Pure transformations**: No side effects
5. **Documentation**: Keep `src/lib/TRANSFORMATIONS.md` updated
